
# Starting factions and pieces. Distances in cells. Every key, in order; a record gives those marked required, and the
# others only when it differs from their default.
#
# A kind of piece:
# kind = {
#   name = example          required
#   icon = Army             default none. Its map icon; a Has_Town's comes from its main building
#   moves = Land            default none, static. The domain it moves in
#   per_turn = 30           default 0. Movement points a turn
#   contact = 9             default 0. Its contact radius
#   contact_on = [Land]     default none. The domains its contact reaches
#   body = 2                default 0. Its body radius
#   hindrance = 3           default 1. Enemy movement cost multiplier inside its contact zone
#   supply = 50             default 0. Supply source value 0..100; 0, not a source
#   slots = 4               default 0. A Has_Town's ordinary slots, until its main building sets them
#   traits = [Captures]     default none. Capturable: changes hands when dissolved by a Captures piece, and the
#                           army inside surrenders. Can_Enter: can enter an empty Can_Contain of its faction.
#                           Fortified: fights sorties from its walls, as does an army inside it. Shares_Supply:
#                           counts in nearby armies' supply shares. Has_Town: has a town economy and a main building,
#                           which gives it its icon, contact, body, hindrance and supply instead of its kind
# }
#
# A faction, with its pieces. Factions play in file order:
# faction = {
#   name = "Example"        required
#   culture = latin         required. By id, from cultures.txt
#   offices = [augustus]    default none. Its own offices, by id from offices.txt
#   capital = "Roma"        required. One of its pieces, by name, where its own offices' armies mobilise
#   color = [190, 80, 70]   required. [r, g, b], 0..255
#   treasury = 30           default 0. Starting coin
#   food_store = 20         default 0. Starting food in its central store
#   piece = { ... }         One per piece:
# }
#
# piece = {
#   kind = settlement            required. By name, from the kinds
#   name = "Example"             default none
#   at = [342, 432]              required. [x, y] from the top left
#   culture = latin              default its faction's
#   capital_of = example_region  default none. The region it is the capital of, by id; at most one per region
#   title = tribune              default none. An army's: the office whose army it is, by id from offices.txt, with
#                                an army_type. Its holder leads it, at 100% readiness, a local one at home where it
#                                starts; its full strength follows its holder's wealth
#   of = "Roma"                  default none. With a building's office, its town
#   men = 3000                   required by an army, or by a settlement's garrison
#   proficiency = 30             default 0. A settlement's garrison's, 0..100
#   foraging = 40                default 40. A settlement's garrison's skill at living off the land, 0..100
#   population = 10              default 0. A Has_Town's starting population
#   buildings = [wheat_fields]   default none. A Has_Town's buildings besides its main, by id, at most its slots
#   main = civitas               required by a Has_Town. Its main building, by id, one with the Main flag
# }

kind = { name = army        icon = Army        moves = Land  per_turn = 30  contact = 9   contact_on = [Land]  body = 2    hindrance = 3                traits = [Captures, Can_Enter, Shares_Supply] }
