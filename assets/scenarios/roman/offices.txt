
# The kinds of office. Factions list their own in pieces.txt; buildings name in buildings.txt the office each creates,
# which takes its town's name: Governor of Ravenna. Every key, in order; an office gives those marked required, and
# the others only when it differs from their default:
#
# office = {
#   id = example               required. A snake_case word the other files name it by
#   name = "Example"           required
#   rank = 2                   required. Its place in the order of offices; whoever is appointed rises to it
#   influence = 15             required. What holding it gives
#   term = 2                   required. Turns its holder serves before he can be promoted or dismissed
#   flags = [Dismissable]      default none. Eternal: held for life. Dismissable: its holder can be dismissed once his
#                              term is served. Feudal: held as a fief; dismissing the holder of one costs every holder
#                              of one in the faction loyalty
#   salary = 8                 default 0. A faction office's coin a turn to its holder, out of the treasury; a
#                              building's office is paid by its building instead
#   income_share = 0.5         default 0. A faction office's share of its faction's income, 0..1, to its holder, out of
#                              the treasury, besides its salary
#   army_type = example_type   default none. The army its holder commands, by id from army_types.txt. Demobilised,
#                              the title gathers its strength; mobilised, beside its town or, for a faction's own
#                              office, its capital, it takes the field
# }
