# The types of army. Titles raise them: the offices in offices.txt name their army types, and their holders' wealth,
# their salaries and shares, size them. Every key, in order; a type gives those marked required, and the others only
# when it differs from their default:
#
# army_type = {
#   id = example                  required. A snake_case word the other files name it by
#   name = "Example"              required
#   scope = Global                required. Local: tied to a home region, the one it is mobilised in. Global: none
#   men_per_wealth = 1000         required. Its full strength, per coin a turn of its holder's wealth
#   proficiency = 50              required. 0..100, at full strength
#   foraging = 40                 default 40. 0..100, at full strength
#   agility = 2                   default 2. 1..4, at full strength
#   traits = []                   default none. By id, from army_traits.txt
#   description = "Example: ..."  required. What it is, shown when it is hovered
# }
