# The traits of armies. Army types name theirs in army_types.txt; an army's modifiers are its traits', added
# together. Every key, in order; a trait gives those marked required, and the others only when it differs from their
# default:
#
# army_trait = {
#   id = example               required. A snake_case word the other files name it by
#   name = "Example"           required
#   agility = 0                default 0. Added to its agility
#   movement = 0               default 0. Movement points added a turn
#   readiness_recovery = 0     default 0. Readiness regained a turn, toward its supply level, when not resting
#   prevent_status = []        default none. Statuses it never takes: Spent, Resting, Seasick
#   description = "Example: ..."  required. What it is, shown when it is hovered
# }
